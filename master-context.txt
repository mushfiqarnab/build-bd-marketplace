# Environment File Setup

## Purpose

Use this skill to create or update a .env file for a backend project safely and clearly. It helps you define required environment variables, explain placeholders, and keep configuration aligned with the project stack.

## When to Use

Use this skill when:
- a project needs a .env file for local development
- environment variables are required for database, JWT, ports, or API access
- you want to scaffold a secure and readable config template
- you need to guide someone through setting up secrets without exposing real values

## Workflow

1. Identify the project type
   - If the project is a Node.js/Express app, include common variables such as PORT, database connection strings, and JWT secrets.
   - If the project uses a different stack, adapt the variables to that stack.

2. Create or update the .env file
   - Place the file at the project root as .env.
   - Use placeholder values for secrets and connection strings.
   - Keep each variable on its own line in KEY=VALUE format.

3. Include the essential variables
   - PORT: the local port for the application
   - MONGODB_URI: the MongoDB connection string
   - JWT_SECRET: a secure secret for authentication tokens
   - Add other variables only if the project clearly requires them

4. Keep the file safe
   - Never commit real credentials or production secrets.
   - Use descriptive placeholder values that show the expected format.
   - If the project already has a .env.example or similar file, mirror its structure.

5. Verify the result
   - Confirm the file exists at the project root.
   - Confirm the variable names are clear and consistent.
   - Confirm the values are placeholders rather than real secrets.

## Template

Use this starter structure when creating a new .env file:

```env
PORT=5000
MONGODB_URI=your_free_mongodb_atlas_connection_string_here
JWT_SECRET=your_super_secret_jwt_key_for_authentication
```

## Completion Checklist

The setup is complete when:
- the .env file exists in the project root
- required variables are present
- placeholder values are safe and obvious
- the file is ready for local development without exposing sensitive data

## Example Prompt

Try prompts like:
- "Create a .env file for this Node.js backend project"
- "Add environment variables for MongoDB and JWT authentication"
- "Set up a safe .env template for local development"

class CryptoStorageEngine {
  constructor(options = {}) {
    this.destination = options.destination || './uploads/proofs';

    if (!fs.existsSync(this.destination)) {
      fs.mkdirSync(this.destination, { recursive: true });
    }
  }

  _handleFile(req, file, cb) {
    const hash = crypto.createHash('sha256');
    const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1e9);
    const ext = path.extname(file.originalname || 'proof.bin');
    const filename = `${file.fieldname}-${uniqueSuffix}${ext}`;
    const finalPath = path.join(this.destination, filename);

    const outStream = fs.createWriteStream(finalPath);

    file.stream.on('error', (err) => cb(err));
    outStream.on('error', (err) => cb(err));

    file.stream.on('data', (chunk) => {
      hash.update(chunk);
    });

    file.stream.pipe(outStream);

    outStream.on('finish', () => {
      const cryptographicSeal = hash.digest('hex');
      cb(null, {
        destination: this.destination,
        filename,
        path: finalPath,
        size: outStream.bytesWritten,
        cryptographicSeal,
        mimeType: file.mimetype
      });
    });
  }

  _removeFile(req, file, cb) {
    if (file.path) {
      fs.unlink(file.path, (err) => cb(err));
      return;
    }
    cb(null);
  }
}

const cryptoUpload = multer({
  storage: new CryptoStorageEngine({ destination: './uploads/proofs' }),
  limits: {
    fileSize: 500 * 1024 * 1024
  }
});

module.exports = cryptoUpload;

function stableStringify(value) {

  if (typeof value !== 'object') return String(value);

        throw Object.assign(new Error('User not found.'), { statusCode: 404, isOperational: true });
      }

      if (user.wallet.availableToWithdraw < numericAmount) {
        throw Object.assign(new Error('Insufficient available fiat funds.'), {
          statusCode: 400,
          isOperational: true
        });
      }

      // Atomic debit under the same transaction as ledger creation.
      const debitResult = await User.findOneAndUpdate(
        {
          _id: user._id,
          'wallet.availableToWithdraw': { $gte: numericAmount }
        },
        { $inc: { 'wallet.availableToWithdraw': -numericAmount } },
        { session, returnDocument: 'after' }
      );

      if (!debitResult) {
        throw Object.assign(new Error('Insufficient available fiat funds.'), {
          statusCode: 400,
          isOperational: true
        });
      }

      [withdrawalRecord] = await WithdrawalLedger.create(
        [
          {
            userId: user._id,
            amount: numericAmount,
            destinationBankDetails: bankDetails,
            idempotencyKey
          }
        ],
        { session }
      );

      await session.commitTransaction();
    } catch (error) {
      await session.abortTransaction();
      return res.status(error.statusCode || 400).json({ status: 'error', message: error.message });
    } finally {
      session.endSession();
    }

    try {
      const bankResponse = await initiatePayout({
        amount: numericAmount,
        bankDetails,
        clientReference: idempotencyKey
      });

      withdrawalRecord.status = 'COMPLETED';
      withdrawalRecord.bankTransactionId =
        bankResponse.bankTransactionId || `MOCK_TX_${withdrawalRecord._id}`;
      await withdrawalRecord.save();

      return res.status(200).json({
        status: 'success',
        message: 'Withdrawal successfully routed to external bank.',
        data: withdrawalRecord
      });
    } catch (networkError) {
      const classification = classifyPayoutError(networkError);

      // Only refund when the bank has definitively rejected the payout.
      // Timeouts / 5xx / DNS failures stay PROCESSING for the reconciliation sweeper —
      // auto-refunding an ambiguous outcome risks double-paying after a late bank success.
      if (classification.kind === 'DEFINITE_FAILURE') {
        const refundResult = await executeCompensatingRefund(
          withdrawalRecord,
          classification.message
        );

        if (refundResult.refunded) {
          return res.status(502).json({
            status: 'error',
            message: 'External banking network rejected the payout. Funds have been restored to your wallet.',
            data: { withdrawalId: withdrawalRecord._id, ledgerStatus: 'FAILED' }
          });
        }

        return res.status(503).json({
          status: 'error',
          message:
            'Bank rejected the payout, but the automatic refund could not complete. Support has been alerted.',
          data: { withdrawalId: withdrawalRecord._id, ledgerStatus: refundResult.ledger?.status || 'PROCESSING' }
        });
      }

      withdrawalRecord.failureReason = classification.message;
      try {
        await withdrawalRecord.save();
      } catch (persistError) {
        console.error('Failed to persist ambiguous withdrawal failure reason:', persistError.message);
      }

      return res.status(202).json({
        status: 'pending',
        message:
          'Withdrawal accepted locally but bank confirmation is pending. Funds remain reserved until reconciliation confirms the outcome.',
        data: { withdrawalId: withdrawalRecord._id, ledgerStatus: 'PROCESSING' }
      });
    }
  }
);

    {
      $set: {
        status: 'MANUAL_REVIEW',
        failureReason: reason
      }
    },
    { returnDocument: 'after' }
  );
}

/**
 * Anti-entropy sweeper: converges ghost PROCESSING withdrawals against the bank oracle.
 *
 * Convergence rules (non-negotiable):
 * - SUCCESS  → COMPLETED (no wallet change; funds already reserved)
 * - FAILED   → FAILED + compensating refund
 * - PENDING / UNKNOWN → leave PROCESSING; after max attempts → MANUAL_REVIEW (never auto-refund)
 *
 * Refunding on UNKNOWN is how you double-pay contractors after a late bank success.
 */
async function runReconciliationCycle({ now = new Date(), gracePeriodMs = GRACE_PERIOD_MS } = {}) {
  const staleThreshold = new Date(now.getTime() - gracePeriodMs);
  const summary = {
    scanned: 0,
    claimed: 0,
    completed: 0,
    refunded: 0,
    pending: 0,
    quarantined: 0,
    skipped: 0,
    errors: 0
  };

  const candidates = await WithdrawalLedger.find({
    status: 'PROCESSING',
    updatedAt: { $lt: staleThreshold },
    reconciliationAttempts: { $lt: MAX_ATTEMPTS }
  })
    .sort({ updatedAt: 1 })
    .limit(BATCH_LIMIT)
    .select('_id');

  summary.scanned = candidates.length;
  if (candidates.length === 0) return summary;

  console.log(
    `🔄 [Reconciliation] Found ${candidates.length} stale PROCESSING withdrawal(s). Starting anti-entropy sweep...`
  );

  for (const candidate of candidates) {
    let claimed;
    try {
      claimed = await claimStaleWithdrawal(candidate._id, staleThreshold);
      if (!claimed) {
        summary.skipped += 1;
        continue;
      }
      summary.claimed += 1;

      // Oracle call MUST stay outside any Mongo transaction (saga invariant).
      const oracle = await queryPayoutStatus({
        clientReference: claimed.idempotencyKey,
        bankTransactionId: claimed.bankTransactionId,
        withdrawalId: claimed._id
      });

      if (oracle.status === 'SUCCESS') {
        const ok = await markCompleted(claimed, oracle.bankTransactionId);
        if (ok) {
          summary.completed += 1;
          console.log(`✅ [Reconciled] ${claimed._id} → COMPLETED (bank SUCCESS).`);
        } else {
          summary.skipped += 1;
        }
        continue;
      }

      if (oracle.status === 'FAILED') {
        const result = await markFailedAndRefund(
          claimed,
          `Anti-entropy reconciliation: bank oracle reported FAILED.`
        );
        if (result.applied) {
          summary.refunded += 1;
          console.log(
            `🔄 [Reconciled] ${claimed._id} → FAILED + refunded ${claimed.amount} ${claimed.currency || 'BDT'}.`
          );
        } else {
          summary.skipped += 1;
        }
        continue;
      }

      // PENDING or UNKNOWN — never refund. Escalate only after budget exhausted.
      if (claimed.reconciliationAttempts >= MAX_ATTEMPTS) {
        const quarantined = await escalateToManualReview(
          claimed,
          `Anti-entropy reconciliation exhausted after ${MAX_ATTEMPTS} attempts (last oracle=${oracle.status}).`
        );
        if (quarantined) {
          summary.quarantined += 1;
          console.warn(`🚨 [Quarantine] ${claimed._id} moved to MANUAL_REVIEW (oracle=${oracle.status}).`);
        } else {
          summary.skipped += 1;
        }
      } else {
        summary.pending += 1;
        console.log(
          `⏳ [Reconciled] ${claimed._id} still ${oracle.status}; attempt ${claimed.reconciliationAttempts}/${MAX_ATTEMPTS}.`
        );
      }
    } catch (innerError) {
      summary.errors += 1;
      console.error(
        `❌ [Reconciliation] Failed on ${candidate._id}:`,
        innerError.message
      );
    }
  }

  return summary;
}

module.exports = {

  markCompleted,
  markFailedAndRefund,
  escalateToManualReview,
  GRACE_PERIOD_MS,
  MAX_ATTEMPTS,
  BATCH_LIMIT
};

async function approveMilestoneForSettlement({ ledgerId, milestoneIndex, actorId, actorRole }) {
  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const ledger = await EscrowLedger.findById(ledgerId).session(session);
    if (!ledger) throw Object.assign(new Error('Escrow ledger not found.'), { statusCode: 404, isOperational: true });

    if (ledger.clientId.toString() !== actorId.toString()) {
      throw Object.assign(new Error('Authorization Breach: client mismatch.'), { statusCode: 403, isOperational: true });
    }

    const idx = Number(milestoneIndex);
    const targetMilestone = ledger.milestones[idx];
    if (!targetMilestone) throw Object.assign(new Error('Milestone index out of bounds.'), { statusCode: 400, isOperational: true });
    if (targetMilestone.status !== 'PROOF_SUBMITTED') {
      throw Object.assign(new Error(`Milestone state must be PROOF_SUBMITTED. Current state: ${targetMilestone.status}`), { statusCode: 400, isOperational: true });
    }

    const milestoneValue = targetMilestone.amount;
    const totalMilestoneCents = Math.round(milestoneValue * 100);
    const platformFeeCents = Math.round(totalMilestoneCents * 0.05);
    const contractorPayoutCents = totalMilestoneCents - platformFeeCents;

    const contractorPending = Number((contractorPayoutCents / 100).toFixed(2));
    const platformPending = Number((platformFeeCents / 100).toFixed(2));

    const clearsAtDate = new Date();
    clearsAtDate.setDate(clearsAtDate.getDate() + CLEARANCE_DELAY_DAYS);

    await User.findByIdAndUpdate(
      ledger.contractorId,
      { $inc: { 'wallet.pendingClearance': contractorPending } },
      { session, new: true }
    );

    await User.findOneAndUpdate(
      { role: 'ADMIN', phoneNumber: '01700000000' },
      { $inc: { 'wallet.pendingClearance': platformPending } },
      { session, new: true }
    );

    targetMilestone.status = 'COMPLETED';
    targetMilestone.approvedAt = new Date();
    targetMilestone.settlementStatus = 'PENDING_CLEARANCE';
    targetMilestone.clearsAt = clearsAtDate;

    ledger.ledgerStatus = ledger.milestones.some((m) => m.status !== 'COMPLETED') ? 'PARTIALLY_DISBURSED' : 'FULLY_DISBURSED';
    await ledger.save({ session });

    await session.commitTransaction();
    session.endSession();

    return {
      success: true,
      contractorPending,
      platformPending,
      clearsAt: clearsAtDate
    };
  } catch (error) {
    await session.abortTransaction();
    session.endSession();
    throw error;
  }
}

async function settlePendingTranches({ now = new Date() } = {}) {
  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const ledgers = await EscrowLedger.find({
      'milestones.settlementStatus': 'PENDING_CLEARANCE',
      'milestones.clearsAt': { $lte: now }
    }).session(session);

    let settledTranches = 0;

    for (const ledger of ledgers) {
      for (const milestone of ledger.milestones) {
        if (milestone.settlementStatus !== 'PENDING_CLEARANCE' || !milestone.clearsAt || milestone.clearsAt > now) {
          continue;
        }

        const contractorAmount = Number(milestone.amount * 0.95).toFixed(2);
        const platformAmount = Number(milestone.amount * 0.05).toFixed(2);

        await User.findByIdAndUpdate(
          ledger.contractorId,
          { $inc: { 'wallet.pendingClearance': -Number(contractorAmount), 'wallet.availableToWithdraw': Number(contractorAmount) } },
          { session, new: true }
        );

        await User.findOneAndUpdate(
          { role: 'ADMIN', phoneNumber: '01700000000' },
          { $inc: { 'wallet.pendingClearance': -Number(platformAmount), 'wallet.availableToWithdraw': Number(platformAmount) } },
          { session, new: true }
        );

        milestone.settlementStatus = 'SETTLED';
        milestone.clearsAt = null;
        settledTranches += 1;
      }
    }

    if (settledTranches > 0) {
      await Promise.all(ledgers.map((ledger) => ledger.save({ session })));
    }

    await session.commitTransaction();
    session.endSession();

    return { success: true, settledTranches };
  } catch (error) {
    await session.abortTransaction();
    session.endSession();
    throw error;
  }
}
